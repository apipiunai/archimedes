import { useState } from "react";

export function Select({ options, value = "", placeholder, setValue, style, emptyOption = false }) {
    return (
        <>
            <style>{`
                select.custom-select {
                    appearance: base-select;
                    padding: 10px;
                    transition: 0.4s;
                    text-wrap: nowrap;
                    cursor: pointer
                }
                
                select.custom-select:hover,
                select.custom-select:focus {
                }
                
                select.custom-select::picker-icon {
                    transition: 0.4s rotate;
                }
                
                select.custom-select:open::picker-icon {
                    rotate: 180deg;
                }
                
                ::picker(select.custom-select) {
                    appearance: base-select;
                    border: none;
                    border-radius: 5px;
                    opacity: 0;
                    transition: all 0.4s allow-discrete;
                    top: calc(anchor(bottom) + 1px);
                    left: anchor(10%);
                }
                
                :open::picker(select.custom-select) {
                    opacity: 1;
                }
                
                @starting-style {
                    :open::picker(select.custom-select) {
                        opacity: 0;
                    }
                }
                
                select.custom-select option {
                    display: flex;
                    justify-content: flex-start;
                    gap: 20px;
                    padding: 10px;
                    transition: 0.4s;
                }
                
                select.custom-select option:checked {
                }
                
                
            `}</style>
            <select style={{...style}} className="custom-select" onChange={(e) => setValue(e.target.value)} value={value}>
                {emptyOption ? <option value="" hidden>{placeholder}</option> : null}
                {options.map((option, index) => {
                    return <option key={index} value={option}>{option}</option>
                })}
            </select>
        </>
    );
}

export default function SelectComponents() {

    const options = ["option1", "option2", "option3"];
    const [value, setValue] = useState(options[0]);

    return (
        <div>
            <Select options={options} value={value} setValue={setValue}  />
            <div style={{marginTop: 20 }}>
                {value}
            </div>
        </div>
    );
}
